     1→'use client'
     2→
     3→import { useState, useCallback, useRef } from 'react'
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { api } from '@/api/client'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Card, CardContent } from '@/components/ui/card'
     9→import { Checkbox } from '@/components/ui/checkbox'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    12→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription } from '@/components/ui/dialog'
    13→import * as XLSX from 'xlsx'
    14→import { DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger } from '@/components/ui/dropdown-menu'
    15→import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle } from '@/components/ui/alert-dialog'
    16→import { Skeleton } from '@/components/ui/skeleton'
    17→import { Upload, Search, Filter, RefreshCw, Trash2, MoreHorizontal, Plus, ArrowRightLeft, Recycle, ChevronLeft, ChevronRight, Phone, Zap, Clock, AlertCircle, Check, Tag, FileSpreadsheet, X } from 'lucide-react'
    18→import { useToast } from '@/hooks/use-toast'
    19→import { STATUS_LABELS, STATUS_COLORS, type PhoneStatus, type PhoneDataItem } from '@/types'
    20→
    21→const PAGE_SIZE = 20
    22→
    23→const STATUS_OPTIONS = [
    24→  { value: '', label: '全部状态' },
    25→  { value: 'pending', label: '未跟进' },
    26→  { value: 'no_answer', label: '无人接听' },
    27→  { value: 'unreachable', label: '打不通' },
    28→  { value: 'connected', label: '已打通' },
    29→]
    30→
    31→export function DataManage() {
    32→  const { toast } = useToast()
    33→  const queryClient = useQueryClient()
    34→
    35→  const [page, setPage] = useState(1)
    36→  const [status, setStatus] = useState<PhoneStatus | ''>('')
    37→  const [keyword, setKeyword] = useState('')
    38→  const [searchInput, setSearchInput] = useState('')
    39→  const [selected, setSelected] = useState<Set<number>>(new Set())
    40→  const [importOpen, setImportOpen] = useState(false)
    41→  const [importFile, setImportFile] = useState<File | null>(null)
    42→  const [importParsed, setImportParsed] = useState<string[]>([])
    43→  const [importInvalid, setImportInvalid] = useState<string[]>([])
    44→  const fileInputRef = useRef<HTMLInputElement>(null)
    45→  const [distributeOpen, setDistributeOpen] = useState(false)
    46→  const [distributeTo, setDistributeTo] = useState('')
    47→  const [batchStatusOpen, setBatchStatusOpen] = useState(false)
    48→  const [_batchStatus, setBatchStatus] = useState<PhoneStatus>('')
    49→  const [deleteOpen, setDeleteOpen] = useState(false)
    50→  const [deleteId, setDeleteId] = useState<number | null>(null)
    51→  const [batchDeleteOpen, setBatchDeleteOpen] = useState(false)
    52→  const [importSource, setImportSource] = useState('')
    53→  const [sourceFilter, setSourceFilter] = useState('')
    54→
    55→  // Fetch data
    56→  const { data, isLoading, isFetching } = useQuery({
    57→    queryKey: ['phoneData', page, status, keyword, sourceFilter],
    58→    queryFn: () => api.getPhoneData({ page, pageSize: PAGE_SIZE, status: status || undefined, keyword: keyword || undefined, source: sourceFilter || undefined }),
    59→  })
    60→
    61→  // Fetch user options for distribute
    62→  const { data: userOptions } = useQuery({
    63→    queryKey: ['userOptions'],
    64→    queryFn: () => api.getUserOptions(),
    65→  })
    66→
    67→  // Fetch source options for filter
    68→  const { data: sourceOptions } = useQuery({
    69→    queryKey: ['sourceOptions'],
    70→    queryFn: () => api.getSourceOptions(),
    71→  })
    72→
    73→  // Stats query (synced with keyword/source filters, single API call)
    74→  const { data: statsData } = useQuery({
    75→    queryKey: ['phoneDataStats', keyword, sourceFilter],
    76→    queryFn: () => api.getPhoneDataStats({ keyword: keyword || undefined, source: sourceFilter || undefined }),
    77→  })
    78→
    79→  // Import mutation
    80→  const importMutation = useMutation({
    81→    mutationFn: ({ phones, source }: { phones: string[]; source?: string }) => api.importPhones(phones, source),
    82→    onSuccess: (res) => {
    83→      toast({ title: '导入成功', description: `成功导入 ${res.data.count} 条数据` })
    84→      setImportOpen(false)
    85→      setImportFile(null)
    86→      setImportParsed([])
    87→      setImportInvalid([])
    88→      setImportSource('')
    89→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
    90→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
    91→      queryClient.invalidateQueries({ queryKey: ['sourceOptions'] })
    92→    },
    93→    onError: (err: Error) => {
    94→      toast({ title: '导入失败', description: err.message })
    95→    },
    96→  })
    97→
    98→  const handleImport = () => {
    99→    importMutation.mutate({ phones: importParsed, source: importSource.trim() || undefined })
   100→  }
   101→
   102→  const handleImportDialogClose = (open: boolean) => {
   103→    setImportOpen(open)
   104→    if (!open) {
   105→      setImportFile(null)
   106→      setImportParsed([])
   107→      setImportInvalid([])
   108→      setImportSource('')
   109→    }
   110→  }
   111→
   112→  // Distribute mutation
   113→  const distributeMutation = useMutation({
   114→    mutationFn: ({ ids, assignedTo }: { ids: number[]; assignedTo: number }) => api.distributePhones(ids, assignedTo),
   115→    onSuccess: () => {
   116→      toast({ title: '分发成功', description: `已分发 ${selected.size} 条数据` })
   117→      setDistributeOpen(false)
   118→      setDistributeTo('')
   119→      setSelected(new Set())
   120→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
   121→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
   122→    },
   123→    onError: (err: Error) => {
   124→      toast({ title: '分发失败', description: err.message })
   125→    },
   126→  })
   127→
   128→  // Auto distribute mutation
   129→  const autoDistributeMutation = useMutation({
   130→    mutationFn: () => api.autoDistribute(),
   131→    onSuccess: () => {
   132→      toast({ title: '自动分发成功', description: '待分发数据已平均分配' })
   133→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
   134→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
   135→    },
   136→    onError: (err: Error) => {
   137→      toast({ title: '自动分发失败', description: err.message })
   138→    },
   139→  })
   140→
   141→  // Recycle mutation
   142→  const recycleMutation = useMutation({
   143→    mutationFn: (ids: number[]) => api.recyclePhones(ids),
   144→    onSuccess: () => {
   145→      toast({ title: '回收成功', description: `已回收 ${selected.size} 条数据` })
   146→      setSelected(new Set())
   147→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
   148→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
   149→    },
   150→    onError: (err: Error) => {
   151→      toast({ title: '回收失败', description: err.message })
   152→    },
   153→  })
   154→
   155→  // Update single status mutation
   156→  const updateStatusMutation = useMutation({
   157→    mutationFn: ({ id, status }: { id: number; status: string }) => api.updatePhoneStatus(id, status),
   158→    onSuccess: () => {
   159→      toast({ title: '状态更新成功' })
   160→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
   161→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
   162→    },
   163→    onError: (err: Error) => {
   164→      toast({ title: '状态更新失败', description: err.message })
   165→    },
   166→  })
   167→
   168→  // Batch status update mutation
   169→  const batchStatusMutation = useMutation({
   170→    mutationFn: ({ ids, status }: { ids: number[]; status: string }) => api.batchUpdateStatus(ids, status),
   171→    onSuccess: () => {
   172→      toast({ title: '批量更新成功', description: `已更新 ${selected.size} 条数据状态` })
   173→      setBatchStatusOpen(false)
   174→      setBatchStatus('pending')
   175→      setSelected(new Set())
   176→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
   177→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
   178→    },
   179→    onError: (err: Error) => {
   180→      toast({ title: '批量更新失败', description: err.message })
   181→    },
   182→  })
   183→
   184→  // Delete mutation
   185→  const deleteMutation = useMutation({
   186→    mutationFn: (id: number) => api.deletePhoneData(id),
   187→    onSuccess: () => {
   188→      toast({ title: '删除成功' })
   189→      setDeleteOpen(false)
   190→      setDeleteId(null)
   191→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
   192→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
   193→    },
   194→    onError: (err: Error) => {
   195→      toast({ title: '删除失败', description: err.message })
   196→    },
   197→  })
   198→
   199→  // Batch delete mutation
   200→  const batchDeleteMutation = useMutation({
   201→    mutationFn: (ids: number[]) => api.batchDeletePhoneData(ids),
   202→    onSuccess: (res) => {
   203→      toast({ title: '批量删除成功', description: `已删除 ${res.data.count} 条数据` })
   204→      setBatchDeleteOpen(false)
   205→      setSelected(new Set())
   206→      queryClient.invalidateQueries({ queryKey: ['phoneData'] })
   207→      queryClient.invalidateQueries({ queryKey: ['phoneDataStats'] })
   208→      queryClient.invalidateQueries({ queryKey: ['sourceOptions'] })
   209→    },
   210→    onError: (err: Error) => {
   211→      toast({ title: '批量删除失败', description: err.message })
   212→    },
   213→  })
   214→
   215→  const handleSearch = () => {
   216→    setKeyword(searchInput)
   217→    setPage(1)
   218→    setSelected(new Set())
   219→  }
   220→
   221→  const handleStatusFilter = (val: string) => {
   222→    setStatus(val === '_all' ? '' : (val as PhoneStatus))
   223→    setPage(1)
   224→    setSelected(new Set())
   225→  }
   226→
   227→  const handleSourceFilter = (val: string) => {
   228→    setSourceFilter(val === '_all' ? '' : val)
   229→    setPage(1)
   230→    setSelected(new Set())
   231→  }
   232→
   233→  const handleReset = () => {
   234→    setStatus('')
   235→    setKeyword('')
   236→    setSearchInput('')
   237→    setSourceFilter('')
   238→    setPage(1)
   239→    setSelected(new Set())
   240→  }
   241→
   242→  const parsePhones = useCallback((phones: (string | number)[]) => {
   243→    const valid: string[] = []
   244→    const invalid: string[] = []
   245→    phones.forEach((raw) => {
   246→      const p = String(raw).trim()
   247→      if (/^1\d{10}$/.test(p)) {
   248→        valid.push(p)
   249→      } else if (p) {
   250→        invalid.push(p)
   251→      }
   252→    })
   253→    setImportParsed(valid)
   254→    setImportInvalid(invalid)
   255→  }, [])
   256→
   257→  const handleFileSelect = useCallback((file: File) => {
   258→    setImportFile(file)
   259→    const reader = new FileReader()
   260→    reader.onload = (e) => {
   261→      try {
   262→        const data = new Uint8Array(e.target!.result as ArrayBuffer)
   263→        const wb = XLSX.read(data, { type: 'array' })
   264→        const ws = wb.Sheets[wb.SheetNames[0]]
   265→        const rows: (string | number)[][] = XLSX.utils.sheet_to_json(ws, { header: 1, raw: true })
   266→        // Extract all values from column A, flatten in case of nested arrays
   267→        const phones = rows.map((row) => row[0]).filter((v) => v !== undefined && v !== null)
   268→        parsePhones(phones)
   269→      } catch {
   270→        setImportParsed([])
   271→        setImportInvalid([])
   272→      }
   273→    }
   274→    reader.readAsArrayBuffer(file)
   275→  }, [parsePhones])
   276→
   277→  const handleDropZoneClick = () => {
   278→    fileInputRef.current?.click()
   279→  }
   280→
   281→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   282→    const file = e.target.files?.[0]
   283→    if (file) handleFileSelect(file)
   284→    e.target.value = ''
   285→  }
   286→
   287→  const handleDragOver = (e: React.DragEvent) => {
   288→    e.preventDefault()
   289→    e.stopPropagation()
   290→  }
   291→
   292→  const handleDrop = (e: React.DragEvent) => {
   293→    e.preventDefault()
   294→    e.stopPropagation()
   295→    const file = e.dataTransfer.files?.[0]
   296→    if (file) handleFileSelect(file)
   297→  }
   298→
   299→  const clearFile = () => {
   300→    setImportFile(null)
   301→    setImportParsed([])
   302→    setImportInvalid([])
   303→  }
   304→
   305→  const toggleSelectAll = () => {
   306→    if (!data?.data.list) return
   307→    const allIds = data.data.list.map((d: PhoneDataItem) => d.id)
   308→    if (selected.size === allIds.length) {
   309→      setSelected(new Set())
   310→    } else {
   311→      setSelected(new Set(allIds))
   312→    }
   313→  }
   314→
   315→  const toggleSelect = (id: number) => {
   316→    const next = new Set(selected)
   317→    if (next.has(id)) {
   318→      next.delete(id)
   319→    } else {
   320→      next.add(id)
   321→    }
   322→    setSelected(next)
   323→  }
   324→
   325→  const totalPages = data?.data ? Math.max(1, Math.ceil(data.data.total / PAGE_SIZE)) : 1
   326→  const items = (data?.data?.list || []) as PhoneDataItem[]
   327→
   328→  const stats = statsData?.data || { total: 0, pending: 0, connected: 0, unreachable: 0 }
   329→
   330→  return (
   331→    <div className="space-y-6">
   332→      {/* Stats cards */}
   333→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   334→        <Card>
   335→          <CardContent className="p-4 flex items-center gap-4">
   336→            <div className="w-10 h-10 rounded-lg bg-primary/10 flex items-center justify-center">
   337→              <Phone className="w-5 h-5 text-primary" />
   338→            </div>
   339→            <div>
   340→              <p className="text-sm text-muted-foreground">总数据</p>
   341→              <p className="text-2xl font-bold">{stats.total}</p>
   342→            </div>
   343→          </CardContent>
   344→        </Card>
   345→        <Card>
   346→          <CardContent className="p-4 flex items-center gap-4">
   347→            <div className="w-10 h-10 rounded-lg bg-slate-100 flex items-center justify-center">
   348→              <Clock className="w-5 h-5 text-slate-600" />
   349→            </div>
   350→            <div>
   351→              <p className="text-sm text-muted-foreground">未跟进</p>
   352→              <p className="text-2xl font-bold">{stats.pending}</p>
   353→            </div>
   354→          </CardContent>
   355→        </Card>
   356→        <Card>
   357→          <CardContent className="p-4 flex items-center gap-4">
   358→            <div className="w-10 h-10 rounded-lg bg-emerald-50 flex items-center justify-center">
   359→              <Check className="w-5 h-5 text-emerald-600" />
   360→            </div>
   361→            <div>
   362→              <p className="text-sm text-muted-foreground">已打通</p>
   363→              <p className="text-2xl font-bold">{stats.connected}</p>
   364→            </div>
   365→          </CardContent>
   366→        </Card>
   367→        <Card>
   368→          <CardContent className="p-4 flex items-center gap-4">
   369→            <div className="w-10 h-10 rounded-lg bg-red-50 flex items-center justify-center">
   370→              <AlertCircle className="w-5 h-5 text-red-600" />
   371→            </div>
   372→            <div>
   373→              <p className="text-sm text-muted-foreground">打不通</p>
   374→              <p className="text-2xl font-bold">{stats.unreachable}</p>
   375→            </div>
   376→          </CardContent>
   377→        </Card>
   378→      </div>
   379→
   380→      {/* Filters */}
   381→      <Card>
   382→        <CardContent className="p-4">
   383→          <div className="flex flex-col sm:flex-row gap-3">
   384→            <Select value={status} onValueChange={handleStatusFilter}>
   385→              <SelectTrigger className="w-full sm:w-36">
   386→                <SelectValue placeholder="全部状态" />
   387→              </SelectTrigger>
   388→              <SelectContent>
   389→                {STATUS_OPTIONS.map((o) => (
   390→                  <SelectItem key={o.value} value={o.value || '_all'}>{o.label}</SelectItem>
   391→                ))}
   392→              </SelectContent>
   393→            </Select>
   394→            <Select value={sourceFilter || '_all'} onValueChange={handleSourceFilter}>
   395→              <SelectTrigger className="w-full sm:w-36">
   396→                <SelectValue placeholder="全部来源" />
   397→              </SelectTrigger>
   398→              <SelectContent>
   399→                <SelectItem value="_all">全部来源</SelectItem>
   400→                {(sourceOptions?.data || []).map((s: string) => (
   401→                  <SelectItem key={s} value={s}>{s}</SelectItem>
   402→                ))}
   403→              </SelectContent>
   404→            </Select>
   405→            <div className="flex-1 flex gap-2">
   406→              <Input
   407→                placeholder="搜索手机号或姓名..."
   408→                value={searchInput}
   409→                onChange={(e) => setSearchInput(e.target.value)}
   410→                onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   411→                className="flex-1"
   412→              />
   413→              <Button variant="outline" onClick={handleSearch} size="default">
   414→                <Search className="w-4 h-4 mr-1.5" />搜索
   415→              </Button>
   416→              <Button variant="ghost" onClick={handleReset} size="default">
   417→                <RefreshCw className="w-4 h-4" />
   418→              </Button>
   419→            </div>
   420→          </div>
   421→        </CardContent>
   422→      </Card>
   423→
   424→      {/* Actions bar */}
   425→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   426→        <div className="flex flex-wrap gap-2">
   427→          <Button onClick={() => { setImportFile(null); setImportParsed([]); setImportInvalid([]); setImportSource(''); setImportOpen(true) }}>
   428→            <Upload className="w-4 h-4 mr-1.5" />导入数据
   429→          </Button>
   430→          <Button variant="outline" onClick={() => autoDistributeMutation.mutate()} disabled={autoDistributeMutation.isPending}>
   431→            <Zap className="w-4 h-4 mr-1.5" />自动分发
   432→          </Button>
   433→          <Button
   434→            variant="outline"
   435→            disabled={selected.size === 0}
   436→            onClick={() => { setDistributeTo(''); setDistributeOpen(true) }}
   437→          >
   438→            <ArrowRightLeft className="w-4 h-4 mr-1.5" />分发
   439→            {selected.size > 0 && <Badge variant="secondary" className="ml-1.5">{selected.size}</Badge>}
   440→          </Button>
   441→          <Button
   442→            variant="outline"
   443→            disabled={selected.size === 0}
   444→            onClick={() => recycleMutation.mutate(Array.from(selected))}
   445→          >
   446→            <Recycle className="w-4 h-4 mr-1.5" />回收
   447→            {selected.size > 0 && <Badge variant="secondary" className="ml-1.5">{selected.size}</Badge>}
   448→          </Button>
   449→          <DropdownMenu open={batchStatusOpen} onOpenChange={setBatchStatusOpen}>
   450→            <DropdownMenuTrigger asChild>
   451→              <Button variant="outline" disabled={selected.size === 0}>
   452→                <Filter className="w-4 h-4 mr-1.5" />批量状态
   453→                {selected.size > 0 && <Badge variant="secondary" className="ml-1.5">{selected.size}</Badge>}
   454→              </Button>
   455→            </DropdownMenuTrigger>
   456→            <DropdownMenuContent align="start">
   457→              {(['pending', 'no_answer', 'unreachable', 'connected'] as PhoneStatus[]).map((s) => (
   458→                <DropdownMenuItem
   459→                  key={s}
   460→                  onClick={() => batchStatusMutation.mutate({ ids: Array.from(selected), status: s })}
   461→                  disabled={batchStatusMutation.isPending}
   462→                >
   463→                  {STATUS_LABELS[s]}
   464→                </DropdownMenuItem>
   465→              ))}
   466→            </DropdownMenuContent>
   467→          </DropdownMenu>
   468→          <Button
   469→            variant="outline"
   470→            disabled={selected.size === 0}
   471→            onClick={() => setBatchDeleteOpen(true)}
   472→            className="text-destructive hover:text-destructive"
   473→          >
   474→            <Trash2 className="w-4 h-4 mr-1.5" />批量删除
   475→            {selected.size > 0 && <Badge variant="secondary" className="ml-1.5">{selected.size}</Badge>}
   476→          </Button>
   477→        </div>
   478→        {isFetching && !isLoading && (
   479→          <span className="text-xs text-muted-foreground flex items-center gap-1.5">
   480→            <RefreshCw className="w-3 h-3 animate-spin" />刷新中...
   481→          </span>
   482→        )}
   483→      </div>
   484→
   485→      {/* Table */}
   486→      <Card>
   487→        <div className="max-h-[calc(100vh-320px)] overflow-y-auto custom-scrollbar">
   488→          <table className="w-full data-table">
   489→            <thead className="sticky top-0 bg-card z-10">
   490→              <tr className="border-b border-border">
   491→                <th className="w-12 px-4 py-3 text-left">
   492→                  <Checkbox
   493→                    checked={items.length > 0 && selected.size === items.length}
   494→                    onCheckedChange={toggleSelectAll}
   495→                  />
   496→                </th>
   497→                <th className="px-4 py-3 text-left text-xs font-medium text-muted-foreground uppercase">手机号</th>
   498→                <th className="px-4 py-3 text-left text-xs font-medium text-muted-foreground uppercase">姓名</th>
   499→                <th className="px-4 py-3 text-left text-xs font-medium text-muted-foreground uppercase">来源</th>
   500→                <th className="px-4 py-3 text-left text-xs font-medium text-muted-foreground uppercase">状态</th>
   501→                <th className="px-4 py-3 text-left text-xs font-medium text-muted-foreground uppercase">分配给</th>
   502→                <th className="px-4 py-3 text-left text-xs font-medium text-muted-foreground uppercase">分配时间</th>
   503→                <th className="px-4 py-3 text-left text-xs font-medium text-muted-foreground uppercase">跟进时间</th>
   504→                <th className="px-4 py-3 text-right text-xs font-medium text-muted-foreground uppercase">操作</th>
   505→              </tr>
   506→            </thead>
   507→            <tbody className="divide-y divide-border">
   508→              {isLoading ? (
   509→                Array.from({ length: 5 }).map((_, i) => (
   510→                  <tr key={i}>
   511→                    <td className="px-4 py-3"><Skeleton className="h-4 w-4" /></td>
   512→                    <td className="px-4 py-3"><Skeleton className="h-4 w-28" /></td>
   513→                    <td className="px-4 py-3"><Skeleton className="h-4 w-16" /></td>
   514→                    <td className="px-4 py-3"><Skeleton className="h-4 w-20" /></td>
   515→                    <td className="px-4 py-3"><Skeleton className="h-5 w-16" /></td>
   516→                    <td className="px-4 py-3"><Skeleton className="h-4 w-16" /></td>
   517→                    <td className="px-4 py-3"><Skeleton className="h-4 w-28" /></td>
   518→                    <td className="px-4 py-3"><Skeleton className="h-4 w-28" /></td>
   519→                    <td className="px-4 py-3"><Skeleton className="h-4 w-16" /></td>

   536→                    </td>

   539→                        className="phone-mono text-sm cursor-pointer select-none hover:text-primary transition-colors"

   546→                      >
   547→                        {item.phone}
   548→                      </span>

   560→                        {STATUS_LABELS[item.status]}
   561→                      </span>
   562→                    </td>
   563→                    <td className="px-4 py-3 text-sm">{item.assignedUser?.nickname || '-'}</td>
   564→                    <td className="px-4 py-3 text-xs text-muted-foreground">
   565→                      {item.assignedAt ? new Date(item.assignedAt).toLocaleString('zh-CN') : '-'}
   566→                    </td>
   567→                    <td className="px-4 py-3 text-xs text-muted-foreground">
   568→                      {item.followedUpAt ? new Date(item.followedUpAt).toLocaleString('zh-CN') : '-'}
   569→                    </td>
   570→                    <td className="px-4 py-3 text-right">
   571→                      <div className="flex items-center justify-end gap-1">
   572→                        <DropdownMenu>
   573→                          <DropdownMenuTrigger asChild>
   574→                            <Button variant="ghost" size="sm" className="h-8 px-2">
   575→                              <Filter className="w-3.5 h-3.5" />
   576→                            </Button>
   577→                          </DropdownMenuTrigger>
   578→                          <DropdownMenuContent align="end">
   579→                            {(['pending', 'no_answer', 'unreachable', 'connected'] as PhoneStatus[]).map((s) => (
   580→                              <DropdownMenuItem
   581→                                key={s}
   582→                                onClick={() => updateStatusMutation.mutate({ id: item.id, status: s })}
   583→                                disabled={updateStatusMutation.isPending}
   584→                              >
   585→                                {STATUS_LABELS[s]}
   586→                              </DropdownMenuItem>
   587→                            ))}
   588→                          </DropdownMenuContent>
   589→                        </DropdownMenu>
   590→                        <Button
   591→                          variant="ghost"
   592→                          size="sm"
   593→                          className="h-8 px-2 text-destructive hover:text-destructive"
   594→                          onClick={() => { setDeleteId(item.id); setDeleteOpen(true) }}
   595→                        >
   596→                          <Trash2 className="w-3.5 h-3.5" />
   597→                        </Button>
   598→                      </div>
   599→                    </td>
   600→                  </tr>
   601→                ))
   602→              )}
   603→            </tbody>
   604→          </table>
   605→        </div>
   606→
   607→        {/* Pagination */}
   608→        {!isLoading && data?.data && (
   609→          <div className="flex items-center justify-between px-4 py-3 border-t border-border">
   610→            <p className="text-sm text-muted-foreground">
   611→              共 {data.data.total} 条，第 {page}/{totalPages} 页
   612→            </p>
   613→            <div className="flex items-center gap-1">
   614→              <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => { setPage(page - 1); setSelected(new Set()) }}>
   615→                <ChevronLeft className="w-4 h-4" />
   616→              </Button>
   617→              {Array.from({ length: Math.min(5, totalPages) }, (_, i) => {
   618→                let pageNum: number
   619→                if (totalPages <= 5) {
   620→                  pageNum = i + 1
   621→                } else if (page <= 3) {
   622→                  pageNum = i + 1
   623→                } else if (page >= totalPages - 2) {
   624→                  pageNum = totalPages - 4 + i
   625→                } else {
   626→                  pageNum = page - 2 + i
   627→                }
   628→                return (
   629→                  <Button
   630→                    key={pageNum}
   631→                    variant={page === pageNum ? 'default' : 'outline'}
   632→                    size="sm"
   633→                    className="w-8 h-8 p-0"
   634→                    onClick={() => { setPage(pageNum); setSelected(new Set()) }}
   635→                  >
   636→                    {pageNum}
   637→                  </Button>
   638→                )
   639→              })}
   640→              <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => { setPage(page + 1); setSelected(new Set()) }}>
   641→                <ChevronRight className="w-4 h-4" />
   642→              </Button>
   643→            </div>
   644→          </div>
   645→        )}
   646→      </Card>
   647→
   648→      {/* Import Dialog */}
   649→      <Dialog open={importOpen} onOpenChange={handleImportDialogClose}>
   650→        <DialogContent className="sm:max-w-lg">
   651→          <DialogHeader>
   652→            <DialogTitle>导入手机号数据</DialogTitle>
   653→            <DialogDescription>上传 Excel 文件（.xlsx / .xls），文件中仅包含手机号一列，无表头。</DialogDescription>
   654→          </DialogHeader>
   655→          <div className="space-y-3">
   656→            <div>
   657→              <label className="text-sm font-medium mb-1.5 block">来源标签</label>
   658→              <Input
   659→                placeholder="请输入数据来源，如：广告投放、老客户推荐..."
   660→                value={importSource}
   661→                onChange={(e) => setImportSource(e.target.value)}
   662→              />
   663→            </div>
   664→            <input
   665→              ref={fileInputRef}
   666→              type="file"
   667→              accept=".xlsx,.xls"
   668→              className="hidden"
   669→              onChange={handleFileInputChange}
   670→            />
   671→            {!importFile ? (
   672→              <div
   673→                className="border-2 border-dashed border-muted-foreground/25 rounded-lg p-8 flex flex-col items-center gap-3 cursor-pointer hover:border-primary/50 hover:bg-primary/5 transition-colors"
   674→                onClick={handleDropZoneClick}
   675→                onDragOver={handleDragOver}
   676→                onDrop={handleDrop}
   677→              >
   678→                <div className="w-12 h-12 rounded-full bg-primary/10 flex items-center justify-center">
   679→                  <FileSpreadsheet className="w-6 h-6 text-primary" />
   680→                </div>
   681→                <div className="text-center">
   682→                  <p className="text-sm font-medium">点击或拖拽上传 Excel 文件</p>
   683→                  <p className="text-xs text-muted-foreground mt-1">支持 .xlsx / .xls 格式</p>
   684→                </div>
   685→              </div>
   686→            ) : (
   687→              <div className="border rounded-lg p-4">
   688→                <div className="flex items-center justify-between">
   689→                  <div className="flex items-center gap-3 min-w-0">
   690→                    <div className="w-10 h-10 rounded-lg bg-emerald-50 flex items-center justify-center shrink-0">
   691→                      <FileSpreadsheet className="w-5 h-5 text-emerald-600" />
   692→                    </div>
   693→                    <div className="min-w-0">
   694→                      <p className="text-sm font-medium truncate">{importFile.name}</p>
   695→                      <p className="text-xs text-muted-foreground">{(importFile.size / 1024).toFixed(1)} KB</p>
   696→                    </div>
   697→                  </div>
   698→                  <Button variant="ghost" size="icon" className="shrink-0 h-8 w-8" onClick={clearFile}>
   699→                    <X className="w-4 h-4" />
   700→                  </Button>
   701→                </div>
   702→              </div>
   703→            )}
   704→            {importParsed.length > 0 && (
   705→              <div className="flex gap-4 text-sm">
   706→                <span className="text-emerald-600 font-medium">有效: {importParsed.length} 条</span>
   707→                {importInvalid.length > 0 && (
   708→                  <span className="text-red-600 font-medium">无效: {importInvalid.length} 条</span>
   709→                )}
   710→              </div>
   711→            )}
   712→          </div>
   713→          <DialogFooter>
   714→            <Button variant="outline" onClick={() => handleImportDialogClose(false)}>取消</Button>
   715→            <Button
   716→              onClick={handleImport}
   717→              disabled={importParsed.length === 0 || importMutation.isPending}
   718→            >
   719→              {importMutation.isPending ? '导入中...' : `确认导入 ${importParsed.length} 条`}
   720→            </Button>
   721→          </DialogFooter>
   722→        </DialogContent>
   723→      </Dialog>
   724→
   725→      {/* Distribute Dialog */}
   726→      <Dialog open={distributeOpen} onOpenChange={setDistributeOpen}>
   727→        <DialogContent className="sm:max-w-sm">
   728→          <DialogHeader>
   729→            <DialogTitle>分发数据</DialogTitle>
   730→            <DialogDescription>将选中的 {selected.size} 条数据分配给指定用户</DialogDescription>
   731→          </DialogHeader>
   732→          <div>
   733→            <Select value={distributeTo} onValueChange={setDistributeTo}>
   734→              <SelectTrigger>
   735→                <SelectValue placeholder="选择用户" />
   736→              </SelectTrigger>
   737→              <SelectContent>
   738→                {userOptions?.data?.map((u) => (
   739→                  <SelectItem key={u.id} value={String(u.id)}>{u.nickname}</SelectItem>
   740→                ))}
   741→              </SelectContent>
   742→            </Select>
   743→          </div>
   744→          <DialogFooter>
   745→            <Button variant="outline" onClick={() => setDistributeOpen(false)}>取消</Button>
   746→            <Button
   747→              onClick={() => distributeMutation.mutate({ ids: Array.from(selected), assignedTo: Number(distributeTo) })}
   748→              disabled={!distributeTo || distributeMutation.isPending}

   750→              {distributeMutation.isPending ? '分发中...' : '确认分发'}

   753→        </DialogContent>
   754→      </Dialog>
   755→
   756→      {/* Delete Confirmation */}
   757→      <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   758→        <AlertDialogContent>
   759→          <AlertDialogHeader>
   760→            <AlertDialogTitle>确认删除</AlertDialogTitle>
   761→            <AlertDialogDescription>删除后数据无法恢复，确定要删除这条数据吗？</AlertDialogDescription>
   762→          </AlertDialogHeader>
   763→          <AlertDialogFooter>
   764→            <AlertDialogCancel>取消</AlertDialogCancel>
   765→            <AlertDialogAction onClick={() => deleteId && deleteMutation.mutate(deleteId)} className="bg-destructive text-white hover:bg-destructive/90">

   768→          </AlertDialogFooter>
   769→        </AlertDialogContent>
   770→      </AlertDialog>
   771→
   772→      {/* Batch Delete Confirmation */}
   773→      <AlertDialog open={batchDeleteOpen} onOpenChange={setBatchDeleteOpen}>
   774→        <AlertDialogContent>
   775→          <AlertDialogHeader>
   776→            <AlertDialogTitle>确认批量删除</AlertDialogTitle>

   778→          </AlertDialogHeader>
   779→          <AlertDialogFooter>
   780→            <AlertDialogCancel>取消</AlertDialogCancel>
   781→            <AlertDialogAction
   782→              onClick={() => batchDeleteMutation.mutate(Array.from(selected))}
   783→              className="bg-destructive text-white hover:bg-destructive/90"
   784→              disabled={batchDeleteMutation.isPending}
   785→            >
   786→              {batchDeleteMutation.isPending ? '删除中...' : '确认删除'}

   788→          </AlertDialogFooter>
   789→        </AlertDialogContent>
